




int test() {
    int a = 1;
    int b = a + 10;
    int c = a + 100;
    int d = a + 1000;
    int e = a + 0;
    int f = 0;
    int sum = 0;
    int i = 0;

    while (i < 12) {
        // 外层 if-else
        if (a < b) {
            a = a + 3;
            e = e + 1;
            // 内层嵌套 if-else
            if (c < d) {
                c = c + 5;
                f = f + 1;
            } else {
                d = d - 2;
                f = f + 2;
            }
        } else {
            b = b - 2;
            e = e + 2;
            if (c > d) {
                c = c - 3;
                f = f + 3;
            } else {
                d = d + 4;
                f = f + 4;
            }
        }

        // 跨分支边界累加: a/b/c/d/e/f 来自不同 BB
        sum = sum + a + b + c + d + e + f;
        a = a + e;
        b = b + f;

        i = i + 1;
    }

	sum = sum + 1;
    return sum;
}
